# 1) Crear la app (Django no acepta rutas con "/")
python manage.py startapp pagos

# 2) Moverla dentro de /apps (en Windows)
move pagos apps\

# Abre config/settings.py y añade: 'apps.pagos' en TENANT_APPS

# 4) Crear migraciones del modelo
python manage.py makemigrations pagos

# 5) Aplicar migraciones en tenants (todos)
python manage.py migrate_schemas --tenant

# (o solo en hotel_tajibo)
python manage.py migrate_schemas --tenant -s hotel_tajibo

SISTEMA DE BACKUPS

# CREAR MIGRACIONES (IMPORTANTE - HACER PRIMERO)
python manage.py makemigrations backups
python manage.py migrate backups

# BACKUPS MANUALES
# Backup completo de toda la base de datos
python manage.py backup_full
python manage.py backup_full --type=manual

# Backup de todos los tenants
python manage.py backup_tenants

# Backup de un tenant específico
python manage.py backup_tenants --schema=hotel_tajibo

# LIMPIEZA DE BACKUPS
# Limpiar backups antiguos (según BACKUP_RETENTION_DAYS del .env)
python manage.py cleanup_backups

# Limpiar solo backups completos
python manage.py cleanup_backups --type=full

# Limpiar solo backups de tenants
python manage.py cleanup_backups --type=tenant

# Limpiar backups más antiguos de 30 días
python manage.py cleanup_backups --older-than=30

# Simular limpieza sin eliminar (ver qué se eliminaría)
python manage.py cleanup_backups --dry-run

# RESTAURACIÓN
# Restaurar un tenant específico
python manage.py restore_tenant hotel_tajibo ruta/al/archivo.sql

# CONFIGURACIÓN EN .env
BACKUP_ENABLED=True              # Habilitar/deshabilitar backups automáticos
BACKUP_FREQUENCY=daily           # daily/weekly/monthly
BACKUP_TIME=03:00               # Hora de ejecución (formato 24h)
BACKUP_RETENTION_DAYS=7         # Días a conservar backups
BACKUP_STORAGE=local            # local/s3/both

# BACKUPS AUTOMÁTICOS
# Los backups automáticos se ejecutan SOLO en producción (DEBUG=False)
# En desarrollo (DEBUG=True) están deshabilitados por seguridad
# El scheduler se inicia automáticamente cuando Django arranca

# Para activar backups automáticos en producción:
# 1. En .env, cambiar: DEBUG=False
# 2. En .env, asegurar: BACKUP_ENABLED=True
# 3. Reiniciar el servidor Django

# ENDPOINTS API (requiere autenticación de admin)
GET    http://localhost:8000/api/backups/                  # Listar backups
GET    http://localhost:8000/api/backups/{id}/             # Detalle de backup
DELETE http://localhost:8000/api/backups/{id}/             # Eliminar backup
GET    http://localhost:8000/api/backups/{id}/descargar/   # Descargar backup
POST   http://localhost:8000/api/backups/crear-full/       # Crear backup completo
POST   http://localhost:8000/api/backups/crear-tenants/    # Crear backups de tenants
GET    http://localhost:8000/api/backups/estadisticas/     # Ver estadísticas
GET    http://localhost:8000/api/backups/configuracion/    # Ver configuración

# UBICACIÓN DE LOS BACKUPS
media/backups/full/      # Backups completos
media/backups/tenant/    # Backups por tenant
